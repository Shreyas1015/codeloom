# codeloom v0.1 — Plan 1: Foundation + Indexer

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `codeloom init` and `codeloom index` work on any repo or folder of repos, writing an AI-free code graph (`graph.json`) and the UI pages, HTTP calls, API routes, DB access, tables and queues found in each repo (`endpoints.json`), incrementally.

**Architecture:** A pnpm monorepo. `packages/core` is Understand Anything's core, vendored (tree-sitter structure, `GraphBuilder`, layers, fingerprints, ignore rules). `packages/cli` is the published `codeloom` package: a typed JSON store, repo discovery, a graph builder on top of core, and framework adapters that read JS/TS files with the TypeScript compiler API. Plan 2 (linker, journeys, MCP) and Plan 3 (web server, dashboard) build on the files this plan writes.

**Tech Stack:** Node ≥ 20, TypeScript 5, pnpm workspaces, Vitest, zod 4, commander 12, `typescript` (compiler API, runtime dependency), `ignore`, vendored `@codeloom/core` (web-tree-sitter).

**Spec:** `docs/superpowers/specs/2026-10-30-codeloom-design.md` (§2 architecture, §3 journeys, §4 data files, §7 testing).

## Global Constraints

- Package name `codeloom`; CLI binary `codeloom`; data folder `.codeloom/`; workspace file `codeloom.workspace.json`.
- Node `>=20`; ESM only (`"type": "module"`); relative imports end in `.js`.
- No AI calls, no network calls, no telemetry anywhere in the package.
- Every JSON file written by codeloom has a `"version"` field, is validated with zod on read, and is written atomically (temp file + rename).
- A bad file is reported with a clear message, never silently overwritten.
- Understand Anything is MIT: keep its copyright notice in `NOTICE`; record upstream commit `9545eb5`.
- Fixtures and tests contain no Shopfront or Acme code; use the generic `demo-shop` fixture.
- Adapters in v0.1: Next.js (app + pages), React Router, generic fetch/axios/ky clients, AdonisJS, Express, NestJS, Fastify, Next.js API routes, Lucid, Knex, Prisma, Drizzle, TypeORM, raw SQL, RabbitMQ (amqplib).
- Decision recorded here (refines spec §4): `.codeloom/fingerprints.json` is a local cache and is git-ignored by `codeloom init`; all other `.codeloom/` files are committed.
- Decision recorded here (refines spec §3): adapters parse JS/TS with the TypeScript compiler API (accurate for TS/TSX, no native build); the code graph itself uses UA's tree-sitter core for every language.
- Ask the author before any `git push`, `npm publish`, or creating a public repo. Commit locally after each task.

`packages/cli/src/store/json.ts`:
```ts
import { existsSync, mkdirSync, readFileSync, renameSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";
import type { z } from "zod";
import { CodeloomError } from "../errors.js";

/** Read and validate a JSON file. Missing file → null. Bad JSON or bad shape → CodeloomError. */
export function readJson<T>(path: string, schema: z.ZodType<T>): T | null {
  if (!existsSync(path)) return null;
  let raw: unknown;
  try {
    raw = JSON.parse(readFileSync(path, "utf8"));
  } catch {
    throw new CodeloomError(
      `${path} is not valid JSON. Delete it and run "codeloom index" again to rebuild it.`,
    );
  }
  const result = schema.safeParse(raw);
  if (!result.success) {
    const lines = result.error.issues.map(
      (i) => `  - ${i.path.join(".") || "(root)"}: ${i.message}`,
    );
    throw new CodeloomError(

import { execFileSync } from "node:child_process";
import { existsSync, readdirSync } from "node:fs";
import { basename, dirname, join, resolve } from "node:path";
import { CodeloomError } from "../errors.js";
import { readJson } from "../store/json.js";
import { WORKSPACE_FILE } from "../store/paths.js";
import { type RepoRef, WorkspaceFileSchema } from "../store/schemas.js";

export function discoverRepos(root: string): RepoRef[] {
  const abs = resolve(root);
  if (existsSync(join(abs, ".git"))) return [{ id: basename(abs), path: "." }];
  return readdirSync(abs, { withFileTypes: true })
    .filter((d) => d.isDirectory() && !d.name.startsWith(".") && existsSync(join(abs, d.name, ".git")))
    .map((d) => ({ id: d.name, path: d.name }))
    .sort((a, b) => a.id.localeCompare(b.id));
}

export function findWorkspaceRoot(start: string): string | null {
  let dir = resolve(start);
  for (;;) {
    if (existsSync(join(dir, WORKSPACE_FILE))) return dir;
    const parent = dirname(dir);
    if (parent === dir) return null;
    dir = parent;
  }
}

function gitToplevel(start: string): string | null {
  try {
    return execFileSync("git", ["rev-parse", "--show-toplevel"], { cwd: start, encoding: "utf8", stdio: "pipe" }).trim();
  } catch {
    return null;
  }
}

export function loadWorkspace(start: string): { root: string; name: string; repos: RepoRef[] } {
  const wsRoot = findWorkspaceRoot(start);
  if (wsRoot) {
    const ws = readJson(join(wsRoot, WORKSPACE_FILE), WorkspaceFileSchema);
    if (ws) return { root: wsRoot, name: ws.name, repos: ws.repos };
  }
  const top = gitToplevel(start);
  if (!top) {
    throw new CodeloomError(
      `No codeloom workspace or git repo found at ${resolve(start)}. Run "codeloom init" in a repo or a folder of repos first.`,
    );
  }
  return { root: top, name: basename(top), repos: [{ id: basename(top), path: "." }] };
}

export function gitHead(repoAbs: string): string {
  try {
    return execFileSync("git", ["rev-parse", "HEAD"], { cwd: repoAbs, encoding: "utf8", stdio: "pipe" }).trim();
  } catch {
    return "uncommitted";
  }
}

import { statSync } from "node:fs";
import { join } from "node:path";
import ignore from "ignore";
import { createIgnoreFilter } from "@codeloom/core";
import { CODELOOM_DIR } from "../store/paths.js";

export function listFiles(repoAbs: string, extraIgnore: string[] = []): string[] {
  const out = execFileSync("git", ["ls-files", "--cached", "--others", "--exclude-standard", "-z"], {
    cwd: repoAbs,
    encoding: "utf8",
    maxBuffer: 256 * 1024 * 1024,
  });

      return false;
    }
  };
  return [...new Set(out.split("\0").filter(Boolean))]
    .filter((rel) => !defaults.isIgnored(rel) && !extra.ignores(rel) && isFile(rel))
    .sort();
}

import ts from "typescript";

export const JS_EXT = /\.(ts|tsx|js|jsx|mjs|cjs|mts|cts)$/;

export function parseSource(file: string, content: string): ts.SourceFile {
  const kind = /\.tsx$/.test(file)
    ? ts.ScriptKind.TSX
    : /\.jsx$/.test(file)
      ? ts.ScriptKind.JSX
      : /\.(ts|mts|cts)$/.test(file)
        ? ts.ScriptKind.TS
        : ts.ScriptKind.JS;
  return ts.createSourceFile(file, content, ts.ScriptTarget.Latest, true, kind);
}

export function walk(node: ts.Node, visit: (n: ts.Node) => void): void {
  visit(node);
  node.forEachChild((child) => walk(child, visit));
}

export function lineOf(node: ts.Node): number {
  const sf = node.getSourceFile();
  return sf.getLineAndCharacterOfPosition(node.getStart(sf)).line + 1;
}

function memberName(name: ts.Node | undefined): string {
  if (!name) return "<member>";
  if (ts.isIdentifier(name) || ts.isPrivateIdentifier(name) || ts.isStringLiteral(name)) return name.text;
  return name.getText();
}

function className(cls: ts.ClassLikeDeclaration): string {
  return cls.name?.text ?? "<anonymous>";
}

export function enclosingSymbol(node: ts.Node): string {
  let cur: ts.Node | undefined = node.parent;
  while (cur) {
    if (ts.isFunctionDeclaration(cur) && cur.name) return cur.name.text;
    if ((ts.isMethodDeclaration(cur) || ts.isGetAccessor(cur) || ts.isSetAccessor(cur)) && ts.isClassLike(cur.parent)) {
      return `${className(cur.parent)}.${memberName(cur.name)}`;
    }
    if ((ts.isArrowFunction(cur) || ts.isFunctionExpression(cur)) && cur.parent) {
      const p = cur.parent;
      if (ts.isVariableDeclaration(p) && ts.isIdentifier(p.name)) return p.name.text;
      if (ts.isPropertyDeclaration(p) && ts.isClassLike(p.parent)) return `${className(p.parent)}.${memberName(p.name)}`;
      if (ts.isFunctionExpression(cur) && cur.name) return cur.name.text;
    }
    cur = cur.parent;
  }
  return "<module>";
}

export function stringValue(expr: ts.Node | undefined): string | undefined {
  if (!expr) return undefined;
  if (ts.isStringLiteral(expr) || ts.isNoSubstitutionTemplateLiteral(expr)) return expr.text;
  return undefined;
}

export function prop(obj: ts.ObjectLiteralExpression, key: string): ts.Expression | undefined {
  for (const p of obj.properties) {
    if (ts.isPropertyAssignment(p) && p.name.getText().replace(/['"]/g, "") === key) return p.initializer;
  }
  return undefined;
}

export function rootName(expr: ts.Expression): string {
  let cur: ts.Expression = expr;
  let afterThis = "";
  for (;;) {
    if (ts.isPropertyAccessExpression(cur)) {
      if (cur.expression.kind === ts.SyntaxKind.ThisKeyword) afterThis = cur.name.text;
      cur = cur.expression;
    } else if (ts.isCallExpression(cur) || ts.isNonNullExpression(cur) || ts.isParenthesizedExpression(cur)) {
      cur = cur.expression;
    } else if (ts.isElementAccessExpression(cur)) {
      cur = cur.expression;
    } else break;
  }
  if (cur.kind === ts.SyntaxKind.ThisKeyword) return afterThis;
  return ts.isIdentifier(cur) ? cur.text : "";
}

export function lastName(expr: ts.Expression): string {
  if (ts.isIdentifier(expr)) return expr.text;
  if (ts.isPropertyAccessExpression(expr)) return expr.name.text;
  if (expr.kind === ts.SyntaxKind.ThisKeyword) return "this";
  return "";
}

const BASE_RE = /(url|uri|api|base|host|origin|endpoint|server|domain|env)/i;
type Part = { text: string } | { expr: ts.Expression };

function collect(expr: ts.Expression, parts: Part[]): boolean {
  if (ts.isParenthesizedExpression(expr)) return collect(expr.expression, parts);
  if (ts.isStringLiteral(expr) || ts.isNoSubstitutionTemplateLiteral(expr)) {
    parts.push({ text: expr.text });
    return true;
  }
  if (ts.isTemplateExpression(expr)) {
    if (expr.head.text) parts.push({ text: expr.head.text });
    for (const span of expr.templateSpans) {
      parts.push({ expr: span.expression });
      if (span.literal.text) parts.push({ text: span.literal.text });
    }
    return true;
  }
  if (ts.isBinaryExpression(expr) && expr.operatorToken.kind === ts.SyntaxKind.PlusToken) {
    return collect(expr.left, parts) && collect(expr.right, parts);
  }
  if (ts.isIdentifier(expr) || ts.isPropertyAccessExpression(expr) || ts.isCallExpression(expr) || ts.isElementAccessExpression(expr)) {
    parts.push({ expr });
    return true;
  }
  return false;
}

function paramName(expr: ts.Expression): string | undefined {
  if (ts.isIdentifier(expr)) return expr.text;
  if (ts.isPropertyAccessExpression(expr)) return expr.name.text;
  if (ts.isCallExpression(expr) && expr.arguments[0]) return paramName(expr.arguments[0]);
  return undefined;
}

export function normalizePath(p: string): string {
  let s = p.trim().replace(/^[a-z][a-z0-9+.-]*:\/\/[^/]+/i, "");
  s = s.split("?")[0]!.split("#")[0]!;
  s = s.replace(/\/{2,}/g, "/");
  if (!s.startsWith("/")) s = `/${s}`;
  if (s.length > 1 && s.endsWith("/")) s = s.slice(0, -1);
  return s;
}

/** Turn a URL expression into a route pattern. Undefined when it is only a variable (nothing literal). */
export function pathPattern(expr: ts.Expression): { path: string; resolved: boolean } | undefined {
  const parts: Part[] = [];
  if (!collect(expr, parts)) return undefined;
  if (!parts.some((p) => "text" in p)) return undefined;
  let out = "";
  let resolved = true;
  parts.forEach((p, i) => {
    if ("text" in p) {
      out += p.text;
      return;
    }
    if (i === 0 && BASE_RE.test(p.expr.getText())) return;
    const name = paramName(p.expr);
    if (name === undefined) {
      resolved = false;
      out += ":param";
    } else out += `:${name}`;
  });
  return { path: normalizePath(out), resolved };
}

export function fileRoute(segments: string[]): string {
  const out = segments
    .filter((s) => !(s.startsWith("(") && s.endsWith(")")) && !s.startsWith("@"))
    .map((s) => {
      const rest = s.match(/^\[\[?\.\.\.(.+?)\]\]?$/);
      if (rest) return `:${rest[1]}*`;
      const dyn = s.match(/^\[(.+)\]$/);
      if (dyn) return `:${dyn[1]}`;
      return s;
    });
  return normalizePath(out.join("/"));
}

export function pascal(s: string): string {
  return s
    .split(/[_\-\s]+/)
    .filter(Boolean)
    .map((w) => w[0]!.toUpperCase() + w.slice(1))
    .join("");
}

export function snakePlural(className: string): string {
  const snake = className.replace(/([a-z0-9])([A-Z])/g, "$1_$2").toLowerCase();
  if (/[^aeiou]y$/.test(snake)) return `${snake.slice(0, -1)}ies`;
  if (/(s|x|z|ch|sh)$/.test(snake)) return `${snake}es`;
  return `${snake}s`;
}

import ts from "typescript";
import type { UiPage } from "../store/schemas.js";
import { fileRoute, lineOf, normalizePath, stringValue, walk } from "./ast.js";
import type { AdapterContext, ParsedFile } from "./types.js";

export function detectPages(file: ParsedFile, ctx: AdapterContext): UiPage[] {
  const out: UiPage[] = [];
  if (ctx.deps.has("next")) {
    const app = file.rel.match(/(?:^|\/)app\/(?:(.*)\/)?page\.(?:tsx|jsx|ts|js)$/);
    if (app) out.push({ route: fileRoute(app[1] ? app[1].split("/") : []), file: file.rel, line: 1, framework: "next-app" });
    const pages = file.rel.match(/(?:^|\/)pages\/(.+)\.(?:tsx|jsx|ts|js)$/);
    if (pages && !/^(api\/|_app$|_document$|_error$)/.test(pages[1]!)) {
      const segs = pages[1]!.split("/");
      if (segs[segs.length - 1] === "index") segs.pop();
      out.push({ route: fileRoute(segs), file: file.rel, line: 1, framework: "next-pages" });
    }
  }
  if ((ctx.deps.has("react-router") || ctx.deps.has("react-router-dom")) && file.sf) {
    out.push(...reactRouterPages(file));
  }
  return out;
}

function jsxRoutePath(node: ts.JsxOpeningLikeElement): string | undefined {
  if (node.tagName.getText() !== "Route") return undefined;
  for (const attr of node.attributes.properties) {
    if (ts.isJsxAttribute(attr) && attr.name.getText() === "path" && attr.initializer && ts.isStringLiteral(attr.initializer)) {
      return attr.initializer.text;
    }
  }
  return undefined;
}

function objectRoutePath(node: ts.ObjectLiteralExpression): string | undefined {
  let path: string | undefined;
  let isRoute = false;
  for (const p of node.properties) {
    if (!ts.isPropertyAssignment(p) && !ts.isShorthandPropertyAssignment(p)) continue;
    const key = p.name.getText();
    if (key === "path" && ts.isPropertyAssignment(p)) path = stringValue(p.initializer);
    if (["element", "Component", "lazy", "children"].includes(key)) isRoute = true;
  }
  return isRoute ? path : undefined;
}

function joinRoute(parent: string, child: string): string {
  return child.startsWith("/") ? child : `${parent.replace(/\/$/, "")}/${child}`;
}

function reactRouterPages(file: ParsedFile): UiPage[] {
  const out: UiPage[] = [];
  walk(file.sf!, (node) => {
    let own: string | undefined;
    if (ts.isJsxOpeningElement(node) || ts.isJsxSelfClosingElement(node)) own = jsxRoutePath(node);
    else if (ts.isObjectLiteralExpression(node)) own = objectRoutePath(node);
    if (own === undefined) return;
    let full = own;
    let cur: ts.Node | undefined = ts.isJsxOpeningElement(node) ? node.parent.parent : node.parent;
    while (cur) {
      let ancestor: string | undefined;
      if (ts.isJsxElement(cur)) ancestor = jsxRoutePath(cur.openingElement);
      else if (ts.isObjectLiteralExpression(cur)) ancestor = objectRoutePath(cur);
      if (ancestor !== undefined) full = joinRoute(ancestor, full);
      cur = cur.parent;
    }
    out.push({ route: normalizePath(full), file: file.rel, line: lineOf(node), framework: "react-router" });
  });
  return out;
}

import ts from "typescript";
import type { HttpCall, HttpMethod } from "../store/schemas.js";
import { enclosingSymbol, lastName, lineOf, pathPattern, prop, stringValue, walk } from "./ast.js";
import type { ParsedFile } from "./types.js";

const METHODS: Record<string, HttpMethod> = { get: "GET", post: "POST", put: "PUT", patch: "PATCH", delete: "DELETE" };
const FETCH_FNS = new Set(["fetch", "$fetch", "ofetch", "useFetch"]);
const CLIENT_RE = /^(axios|ky|got|superagent)$|(api|client|http|request|instance|fetcher)$/i;

function methodOf(value: string | undefined): HttpMethod | undefined {
  return value ? METHODS[value.toLowerCase()] : undefined;
}

export function detectHttpCalls(file: ParsedFile): HttpCall[] {
  if (!file.sf) return [];
  const out: HttpCall[] = [];
  walk(file.sf, (node) => {
    if (!ts.isCallExpression(node)) return;
    const callee = node.expression;
    const [first, second] = node.arguments;
    let method: HttpMethod | undefined;
    let urlExpr: ts.Expression | undefined;
    let client = "";

    if (ts.isIdentifier(callee) && FETCH_FNS.has(callee.text) && first) {
      const init = second && ts.isObjectLiteralExpression(second) ? second : undefined;
      method = (init && methodOf(stringValue(prop(init, "method")))) ?? "GET";
      urlExpr = first;
      client = callee.text;
    } else if (ts.isIdentifier(callee) && callee.text === "axios" && first && ts.isObjectLiteralExpression(first)) {
      urlExpr = prop(first, "url");
      method = methodOf(stringValue(prop(first, "method"))) ?? "GET";
      client = "axios";
    } else if (ts.isPropertyAccessExpression(callee) && METHODS[callee.name.text] && first) {
      const receiver = lastName(callee.expression);
      if (receiver && receiver !== "this" && CLIENT_RE.test(receiver)) {
        method = METHODS[callee.name.text];
        urlExpr = first;
        client = receiver;
      }
    }
    if (!method || !urlExpr) return;
    const pattern = pathPattern(urlExpr);
    out.push({
      method,
      path: pattern?.path ?? urlExpr.getText(),
      raw: urlExpr.getText(),
      resolved: pattern?.resolved ?? false,
      client,
      at: { file: file.rel, line: lineOf(node), symbol: enclosingSymbol(node) },
    });
  });
  return out;
}

import ts from "typescript";
import type { HttpMethod, Route } from "../store/schemas.js";
import { enclosingSymbol, fileRoute, lineOf, normalizePath, pascal, pathPattern, prop, stringValue, walk } from "./ast.js";
import type { AdapterContext, ParsedFile } from "./types.js";

const ROUTE_METHODS: Record<string, HttpMethod> = {
  get: "GET", post: "POST", put: "PUT", patch: "PATCH", delete: "DELETE", del: "DELETE", any: "ANY", all: "ANY",
};
const ROUTER_NAMES = ["router", "route", "routes", "Route", "app", "server", "fastify", "r"];
const ROUTER_FACTORIES = /^(express|express\.Router|Router|fastify|Fastify)$/;
const BACKEND_DEPS = ["@adonisjs/core", "express", "fastify", "@nestjs/core", "next"];
type Framework = Route["framework"];

export function adonisStringHandler(s: string): string {
  const last = s.split("/").pop()!;
  const dot = last.lastIndexOf(".");
  const file = dot >= 0 ? last.slice(0, dot) : last;
  const method = dot >= 0 ? last.slice(dot + 1) : "handle";
  const cls = /_/.test(file) || /^[a-z]/.test(file) ? pascal(file) : file;
  return `${cls}.${method}`;
}

function frameworkOf(ctx: AdapterContext): Framework {
  if (ctx.deps.has("@adonisjs/core")) return "adonis";
  if (ctx.deps.has("fastify")) return "fastify";
  return "express";
}

function handlerOf(h: ts.Expression | undefined, label: string): { handler: string; inline: boolean } {
  if (!h || ts.isArrowFunction(h) || ts.isFunctionExpression(h)) return { handler: label, inline: true };
  if (ts.isArrayLiteralExpression(h)) {
    const [cls, method] = h.elements;
    const name = cls ? (ts.isIdentifier(cls) ? cls.text : cls.getText()) : "<unknown>";
    return { handler: `${name}.${stringValue(method) ?? "handle"}`, inline: false };
  }
  const s = stringValue(h);
  if (s !== undefined) return { handler: adonisStringHandler(s), inline: false };
  if (ts.isCallExpression(h)) return handlerOf(h.arguments[h.arguments.length - 1], label);
  return { handler: h.getText(), inline: false };
}

function groupPrefix(node: ts.Node): string {
  const prefixes: string[] = [];
  let cur: ts.Node | undefined = node.parent;
  while (cur) {
    if (
      (ts.isArrowFunction(cur) || ts.isFunctionExpression(cur)) &&
      ts.isCallExpression(cur.parent) &&
      ts.isPropertyAccessExpression(cur.parent.expression) &&
      cur.parent.expression.name.text === "group"
    ) {
      let chain: ts.Node = cur.parent;
      while (ts.isPropertyAccessExpression(chain.parent) && ts.isCallExpression(chain.parent.parent)) {
        const call = chain.parent.parent;
        if (chain.parent.name.text === "prefix") {
          const s = stringValue(call.arguments[0]);
          if (s) prefixes.unshift(s);
        }
        chain = call;
      }
    }
    cur = cur.parent;
  }
  return prefixes.join("/");
}

function routerCall(node: ts.CallExpression, routers: Set<string>, file: ParsedFile, framework: Framework): Route | undefined {
  if (!ts.isPropertyAccessExpression(node.expression)) return undefined;
  const name = node.expression.name.text;
  const recv = node.expression.expression;
  const at = { file: file.rel, line: lineOf(node), symbol: enclosingSymbol(node) };

  if (name === "route" && ts.isIdentifier(recv) && routers.has(recv.text) && node.arguments[0] && ts.isObjectLiteralExpression(node.arguments[0])) {
    const o = node.arguments[0];
    const m = stringValue(prop(o, "method"));
    const url = prop(o, "url") ?? prop(o, "path");
    const p = url && pathPattern(url);
    if (!m || !p) return undefined;
    const method = ROUTE_METHODS[m.toLowerCase()] ?? "ANY";
    return { method, path: p.path, ...handlerOf(prop(o, "handler"), `${method} ${p.path}`), framework: "fastify", at };
  }

  const method = ROUTE_METHODS[name];
  if (!method) return undefined;
  let path: string | undefined;
  if (ts.isIdentifier(recv) && routers.has(recv.text)) {
    if (node.arguments.length < 2) return undefined;
    path = pathPattern(node.arguments[0]!)?.path;
  } else if (
    ts.isCallExpression(recv) &&
    ts.isPropertyAccessExpression(recv.expression) &&
    recv.expression.name.text === "route" &&
    ts.isIdentifier(recv.expression.expression) &&
    routers.has(recv.expression.expression.text)
  ) {
    const s = stringValue(recv.arguments[0]);
    path = s === undefined ? undefined : normalizePath(s);
  } else return undefined;
  if (path === undefined) return undefined;
  const full = normalizePath(`${groupPrefix(node)}/${path}`);
  return {
    method,
    path: full,
    ...handlerOf(node.arguments[node.arguments.length - 1], `${method} ${full}`),
    framework,
    at,
  };
}

function decoratorCall(node: ts.Node, names: string[]): { name: string; arg?: ts.Expression } | undefined {
  const decorators = ts.canHaveDecorators(node) ? (ts.getDecorators(node) ?? []) : [];
  for (const d of decorators) {
    if (ts.isCallExpression(d.expression) && ts.isIdentifier(d.expression.expression) && names.includes(d.expression.expression.text)) {
      return { name: d.expression.expression.text, arg: d.expression.arguments[0] };
    }
  }
  return undefined;
}

function nestRoutes(cls: ts.ClassDeclaration, file: ParsedFile): Route[] {
  const controller = decoratorCall(cls, ["Controller"]);
  if (!controller) return [];
  let base = "";
  if (controller.arg) {
    base = stringValue(controller.arg) ?? (ts.isObjectLiteralExpression(controller.arg) ? (stringValue(prop(controller.arg, "path")) ?? "") : "");
  }
  const clsName = cls.name?.text ?? "<anonymous>";
  const out: Route[] = [];
  for (const member of cls.members) {
    if (!ts.isMethodDeclaration(member)) continue;
    const d = decoratorCall(member, ["Get", "Post", "Put", "Patch", "Delete", "All"]);
    if (!d) continue;
    const method: HttpMethod = d.name === "All" ? "ANY" : (d.name.toUpperCase() as HttpMethod);
    const handler = `${clsName}.${member.name.getText()}`;
    out.push({
      method,
      path: normalizePath(`${base}/${d.arg ? (stringValue(d.arg) ?? "") : ""}`),
      handler,
      inline: false,
      framework: "nest",
      at: { file: file.rel, line: lineOf(member), symbol: handler },
    });
  }
  return out;
}

function nextApiRoutes(file: ParsedFile): Route[] | undefined {
  const app = file.rel.match(/(?:^|\/)app\/(?:(.*)\/)?route\.(?:ts|js)$/);
  if (app && file.sf) {
    const path = fileRoute(app[1] ? app[1].split("/") : []);
    const out: Route[] = [];
    for (const st of file.sf.statements) {
      const exported = ts.canHaveModifiers(st) && (ts.getModifiers(st) ?? []).some((m) => m.kind === ts.SyntaxKind.ExportKeyword);
      if (!exported) continue;
      const names: string[] = [];
      if (ts.isFunctionDeclaration(st) && st.name) names.push(st.name.text);
      if (ts.isVariableStatement(st)) {
        for (const d of st.declarationList.declarations) if (ts.isIdentifier(d.name)) names.push(d.name.text);
      }
      for (const n of names) {
        if (["GET", "POST", "PUT", "PATCH", "DELETE"].includes(n)) {
          out.push({ method: n as HttpMethod, path, handler: n, inline: false, framework: "next-api", at: { file: file.rel, line: lineOf(st), symbol: n } });
        }
      }
    }
    return out;
  }
  const pages = file.rel.match(/(?:^|\/)pages\/api\/(.+)\.(?:ts|js)$/);
  if (pages) {
    const segs = pages[1]!.split("/");
    if (segs[segs.length - 1] === "index") segs.pop();
    return [{ method: "ANY", path: fileRoute(["api", ...segs]), handler: "default", inline: false, framework: "next-api", at: { file: file.rel, line: 1, symbol: "default" } }];
  }
  return undefined;
}

export function detectRoutes(file: ParsedFile, ctx: AdapterContext): Route[] {
  if (!BACKEND_DEPS.some((d) => ctx.deps.has(d))) return [];
  if (ctx.deps.has("next")) {
    const next = nextApiRoutes(file);
    if (next) return next;
  }
  if (!file.sf) return [];
  const routers = new Set(ROUTER_NAMES);
  walk(file.sf, (n) => {
    if (ts.isVariableDeclaration(n) && ts.isIdentifier(n.name) && n.initializer && ts.isCallExpression(n.initializer) && ROUTER_FACTORIES.test(n.initializer.expression.getText())) {
      routers.add(n.name.text);
    }
  });
  const framework = frameworkOf(ctx);
  const out: Route[] = [];
  walk(file.sf, (node) => {
    if (ts.isCallExpression(node)) {
      const route = routerCall(node, routers, file, framework);
      if (route) out.push(route);
    }
    if (ts.isClassDeclaration(node) && ctx.deps.has("@nestjs/core")) out.push(...nestRoutes(node, file));
  });
  return out;
}

import ts from "typescript";
import type { DbAccess, DbOp, ModelDef, TableDef } from "../store/schemas.js";
import { enclosingSymbol, lineOf, prop, rootName, snakePlural, stringValue, walk } from "./ast.js";
import type { AdapterContext, EndpointsPart, ParsedFile } from "./types.js";

type DbPart = Pick<EndpointsPart, "dbAccess" | "tables" | "models">;

const DB_DEPS = ["@adonisjs/lucid", "knex", "@prisma/client", "drizzle-orm", "typeorm", "pg", "mysql2", "better-sqlite3", "postgres", "sqlite3"];
const QB_ROOTS = /^(db|knex|trx|database|Database|connection|query|qb|sql|tx)$/;
const MODEL_STATIC: Record<string, DbOp> = {
  create: "insert", createMany: "insert", firstOrCreate: "insert", fetchOrCreateMany: "insert",
  updateOrCreate: "update", updateOrCreateMany: "update",
  find: "read", findOrFail: "read", findBy: "read", findByOrFail: "read", findMany: "read", findManyBy: "read",
  first: "read", firstOrFail: "read", all: "read", query: "read", truncate: "delete",
};
const PRISMA_OPS: Record<string, DbOp> = {
  findMany: "read", findUnique: "read", findUniqueOrThrow: "read", findFirst: "read", findFirstOrThrow: "read",
  count: "read", aggregate: "read", groupBy: "read",
  create: "insert", createMany: "insert", createManyAndReturn: "insert",
  update: "update", updateMany: "update", upsert: "update", delete: "delete", deleteMany: "delete",
};
const REPO_OPS: Record<string, DbOp> = {
  find: "read", findOne: "read", findOneBy: "read", findBy: "read", findOneOrFail: "read", count: "read",
  save: "insert", insert: "insert", create: "insert", update: "update", upsert: "update",
  delete: "delete", remove: "delete", softDelete: "delete",
};

export function sqlTables(text: string): Array<{ table: string; op: DbOp }> {
  if (!/^\s*(select|insert|update|delete|with)\b/i.test(text)) return [];
  const out: Array<{ table: string; op: DbOp }> = [];
  const seen = new Set<string>();
  const add = (raw: string, op: DbOp) => {
    const name = raw.replace(/["`[\]]/g, "").split(".").pop()!.toLowerCase();
    if (!/^[a-z_][a-z0-9_]*$/.test(name) || seen.has(`${op}:${name}`)) return;
    seen.add(`${op}:${name}`);
    out.push({ table: name, op });
  };
  const ident = `([\\w."\`\\[\\]]+)`;
  for (const m of text.matchAll(new RegExp(`insert\\s+into\\s+${ident}`, "gi"))) add(m[1]!, "insert");
  for (const m of text.matchAll(new RegExp(`(?:^|;)\\s*update\\s+${ident}\\s+set`, "gi"))) add(m[1]!, "update");
  for (const m of text.matchAll(new RegExp(`delete\\s+from\\s+${ident}`, "gi"))) add(m[1]!, "delete");
  for (const m of text.matchAll(new RegExp(`(?<!delete\\s+)\\b(?:from|join)\\s+${ident}`, "gi"))) add(m[1]!, "read");
  return out;
}

function opFromChain(call: ts.CallExpression): DbOp {
  let cur: ts.Node = call;
  while (ts.isPropertyAccessExpression(cur.parent) && ts.isCallExpression(cur.parent.parent)) {
    const name = cur.parent.name.text;
    if (/^(insert|insertMany|createMany|multiInsert)$/.test(name)) return "insert";
    if (/^(update|increment|decrement|merge)$/.test(name)) return "update";
    if (/^(delete|del|truncate)$/.test(name)) return "delete";
    cur = cur.parent.parent;
  }
  return "read";
}

function importedFrom(sf: ts.SourceFile, re: RegExp): Set<string> {
  const names = new Set<string>();
  for (const st of sf.statements) {
    if (!ts.isImportDeclaration(st) || !ts.isStringLiteral(st.moduleSpecifier) || !re.test(st.moduleSpecifier.text)) continue;
    const clause = st.importClause;
    if (clause?.name) names.add(clause.name.text);
    if (clause?.namedBindings && ts.isNamedImports(clause.namedBindings)) {
      for (const el of clause.namedBindings.elements) names.add(el.name.text);
    }
  }
  return names;
}

function classTableName(node: ts.Node, arg: ts.Expression): string | undefined {
  if (!ts.isPropertyAccessExpression(arg) || arg.expression.kind !== ts.SyntaxKind.ThisKeyword) return undefined;
  let cur: ts.Node | undefined = node.parent;
  while (cur && !ts.isClassLike(cur)) cur = cur.parent;
  if (!cur || !ts.isClassLike(cur)) return undefined;
  for (const m of cur.members) {
    if (ts.isPropertyDeclaration(m) && m.name.getText() === arg.name.text) return stringValue(m.initializer);
  }
  return undefined;
}

function modelClass(cls: ts.ClassDeclaration, rel: string): ModelDef | undefined {
  const name = cls.name?.text;
  if (!name) return undefined;
  const heritage = (cls.heritageClauses ?? []).map((h) => h.getText()).join(" ");
  if (/\bBaseModel\b/.test(heritage)) {
    let table = snakePlural(name);
    for (const m of cls.members) {
      const isStatic = ts.canHaveModifiers(m) && (ts.getModifiers(m) ?? []).some((x) => x.kind === ts.SyntaxKind.StaticKeyword);
      if (ts.isPropertyDeclaration(m) && isStatic && m.name.getText() === "table") table = stringValue(m.initializer) ?? table;
    }
    return { model: name, table, file: rel, source: "lucid" };
  }
  const decorators = ts.canHaveDecorators(cls) ? (ts.getDecorators(cls) ?? []) : [];
  for (const d of decorators) {
    if (ts.isCallExpression(d.expression) && d.expression.expression.getText() === "Entity") {
      const arg = d.expression.arguments[0];
      const fromObj = arg && ts.isObjectLiteralExpression(arg) ? stringValue(prop(arg, "name")) : undefined;
      const table = stringValue(arg) ?? fromObj ?? name.replace(/([a-z0-9])([A-Z])/g, "$1_$2").toLowerCase();
      return { model: name, table, file: rel, source: "typeorm" };
    }
  }
  return undefined;
}

function prismaSchema(file: ParsedFile): DbPart {
  const tables: TableDef[] = [];
  const models: ModelDef[] = [];
  for (const m of file.content.matchAll(/^model\s+(\w+)\s*\{([\s\S]*?)^\}/gm)) {
    const mapped = m[2]!.match(/@@map\(\s*"([^"]+)"\s*\)/);
    const table = mapped ? mapped[1]! : m[1]!;
    const line = file.content.slice(0, m.index).split("\n").length;
    tables.push({ name: table, file: file.rel, line, source: "prisma" });
    models.push({ model: m[1]!, table, file: file.rel, source: "prisma" });
  }
  return { dbAccess: [], tables, models };
}

function sqlFile(file: ParsedFile): DbPart {
  const tables: TableDef[] = [];
  for (const m of file.content.matchAll(/create\s+table\s+(?:if\s+not\s+exists\s+)?([\w."`]+)/gi)) {
    const name = m[1]!.replace(/["`]/g, "").split(".").pop()!;
    tables.push({ name, file: file.rel, line: file.content.slice(0, m.index).split("\n").length, source: "migration" });
  }
  return { dbAccess: [], tables, models: [] };
}

export function detectDb(file: ParsedFile, ctx: AdapterContext): DbPart {
  if (file.rel.endsWith(".prisma")) return prismaSchema(file);
  if (file.rel.endsWith(".sql")) return sqlFile(file);
  const res: DbPart = { dbAccess: [], tables: [], models: [] };
  if (!file.sf || !DB_DEPS.some((d) => ctx.deps.has(d))) return res;
  const modelImports = importedFrom(file.sf, /model/i);
  const drizzle = ctx.deps.has("drizzle-orm");
  const at = (n: ts.Node) => ({ file: file.rel, line: lineOf(n), symbol: enclosingSymbol(n) });
  const push = (a: DbAccess) => res.dbAccess.push(a);

  walk(file.sf, (node) => {
    if (ts.isClassDeclaration(node)) {
      const m = modelClass(node, file.rel);
      if (m) res.models.push(m);
    }
    if (ts.isStringLiteral(node) || ts.isNoSubstitutionTemplateLiteral(node) || ts.isTemplateExpression(node)) {
      const text = ts.isTemplateExpression(node) ? node.getText().slice(1, -1) : node.text;
      for (const t of sqlTables(text)) push({ table: t.table, op: t.op, via: "sql", at: at(node) });
    }
    if (!ts.isCallExpression(node)) return;
    const e = node.expression;
    const arg0 = node.arguments[0];

    if (ts.isPropertyAccessExpression(e) && ts.isIdentifier(e.expression) && modelImports.has(e.expression.text) && MODEL_STATIC[e.name.text]) {
      const op = e.name.text === "query" ? opFromChain(node) : MODEL_STATIC[e.name.text]!;
      push({ model: e.expression.text, op, via: "lucid", at: at(node) });
      return;
    }
    if (ctx.deps.has("@prisma/client") && ts.isPropertyAccessExpression(e) && PRISMA_OPS[e.name.text] && ts.isPropertyAccessExpression(e.expression) && /prisma|db|client|tx/i.test(rootName(e.expression))) {
      push({ model: e.expression.name.text, op: PRISMA_OPS[e.name.text]!, via: "prisma", at: at(node) });
      return;
    }
    if (ts.isPropertyAccessExpression(e) && REPO_OPS[e.name.text] && ts.isCallExpression(e.expression) && /getRepository$/.test(e.expression.expression.getText())) {
      const model = e.expression.arguments[0];
      if (model && ts.isIdentifier(model)) push({ model: model.text, op: REPO_OPS[e.name.text]!, via: "typeorm", at: at(node) });
      return;
    }
    if (drizzle && ts.isPropertyAccessExpression(e) && ["insert", "update", "delete"].includes(e.name.text) && arg0 && ts.isIdentifier(arg0)) {
      push({ model: arg0.text, op: e.name.text as DbOp, via: "drizzle", at: at(node) });
      return;
    }
    if (ts.isPropertyAccessExpression(e) && ["from", "table", "into"].includes(e.name.text) && arg0) {
      const table = stringValue(arg0);
      if (table && (QB_ROOTS.test(rootName(e.expression)) || ts.isCallExpression(e.expression))) {
        push({ table, op: opFromChain(node), via: "query-builder", at: at(node) });
        return;
      }
      if (!table && drizzle && ts.isIdentifier(arg0)) {
        push({ model: arg0.text, op: "read", via: "drizzle", at: at(node) });
        return;
      }
    }
    if (ts.isIdentifier(e) && /^(knex|db|trx)$/.test(e.text) && stringValue(arg0)) {
      push({ table: stringValue(arg0)!, op: opFromChain(node), via: "query-builder", at: at(node) });
      return;
    }
    if (ts.isPropertyAccessExpression(e) && /^createTable(IfNotExists)?$/.test(e.name.text) && arg0) {
      const name = stringValue(arg0) ?? classTableName(node, arg0);
      if (name) res.tables.push({ name, file: file.rel, line: lineOf(node), source: "migration" });
      return;
    }
    if (ts.isIdentifier(e) && /^(pgTable|mysqlTable|sqliteTable)$/.test(e.text) && stringValue(arg0) && ts.isVariableDeclaration(node.parent) && ts.isIdentifier(node.parent.name)) {
      const table = stringValue(arg0)!;
      res.tables.push({ name: table, file: file.rel, line: lineOf(node), source: "drizzle" });
      res.models.push({ model: node.parent.name.text, table, file: file.rel, source: "drizzle" });
    }
  });
  return res;
}

import ts from "typescript";
import type { QueueOp } from "../store/schemas.js";
import { enclosingSymbol, lastName, lineOf, stringValue, walk } from "./ast.js";
import type { AdapterContext, ParsedFile } from "./types.js";

const RECEIVER_RE = /(channel|^ch$|amqp|rabbit|broker|queue|mq|publisher|consumer)/i;

export function detectQueues(file: ParsedFile, ctx: AdapterContext): QueueOp[] {
  if (!file.sf || ![...ctx.deps].some((d) => /amqp|rabbit/i.test(d))) return [];
  const out: QueueOp[] = [];
  walk(file.sf, (node) => {
    if (!ts.isCallExpression(node) || !ts.isPropertyAccessExpression(node.expression)) return;
    if (!RECEIVER_RE.test(lastName(node.expression.expression))) return;
    const at = { file: file.rel, line: lineOf(node), symbol: enclosingSymbol(node) };
    const [a0, a1] = node.arguments;
    switch (node.expression.name.text) {
      case "sendToQueue": {
        const q = stringValue(a0);
        if (q) out.push({ queue: q, op: "publish", at });
        break;
      }
      case "publish": {
        const exchange = stringValue(a0);
        const key = stringValue(a1);
        if (exchange !== undefined && key) out.push({ queue: exchange ? `${exchange}:${key}` : key, op: "publish", at });
        break;
      }
      case "consume": {
        const q = stringValue(a0);
        if (q) out.push({ queue: q, op: "consume", at });
        break;
      }
    }
  });
  return out;
}

export function loadContext(repoAbs: string, files: string[]): AdapterContext {
  const deps = new Set<string>();
  for (const rel of files) {
    if (rel !== "package.json" && !rel.endsWith("/package.json")) continue;
    try {
      const pkg = JSON.parse(readFileSync(join(repoAbs, rel), "utf8")) as Record<string, Record<string, string> | undefined>;
      for (const key of ["dependencies", "devDependencies", "peerDependencies"]) {
        for (const name of Object.keys(pkg[key] ?? {})) deps.add(name);
      }
    } catch {
      // A broken package.json only means fewer adapters apply.
    }
  }
  return { deps };
}

export function runAdapters(rel: string, content: string, ctx: AdapterContext): EndpointsPart {
  const part = emptyPart();
  const isJs = JS_EXT.test(rel) && !rel.endsWith(".d.ts");
  if (!isJs && !rel.endsWith(".prisma") && !rel.endsWith(".sql")) return part;
  const file: ParsedFile = { rel, content, sf: isJs ? parseSource(rel, content) : undefined };
  part.pages.push(...detectPages(file, ctx));
  part.httpCalls.push(...detectHttpCalls(file));
  part.routes.push(...detectRoutes(file, ctx));
  const db = detectDb(file, ctx);
  part.dbAccess.push(...db.dbAccess);
  part.tables.push(...db.tables);
  part.models.push(...db.models);
  part.queues.push(...detectQueues(file, ctx));
  return part;
}

const byLoc = <T extends { at: { file: string; line: number } }>(a: T, b: T) =>
  a.at.file.localeCompare(b.at.file) || a.at.line - b.at.line;

export function finalizeEndpoints(repo: string, parts: EndpointsPart[]): EndpointsFile {
  const all = emptyPart();
  for (const p of parts) {
    for (const key of Object.keys(all) as Array<keyof EndpointsPart>) {
      (all[key] as unknown[]).push(...(p[key] as unknown[]));
    }
  }
  const exact = new Map(all.models.map((m) => [m.model, m.table]));
  const lower = new Map(all.models.map((m) => [m.model.toLowerCase(), m.table]));
  const dbAccess = all.dbAccess.map((a) => {
    if (a.table || !a.model) return a;
    const table = exact.get(a.model) ?? lower.get(a.model.toLowerCase());
    return table ? { ...a, table } : a;
  });
  const tables = new Map<string, TableDef>();
  for (const t of all.tables) if (!tables.has(t.name)) tables.set(t.name, t);
  for (const m of all.models) {
    if (!tables.has(m.table)) tables.set(m.table, { name: m.table, file: m.file, line: 1, source: "model" });
  }
  return {
    version: 1,
    repo,
    generatedAt: new Date().toISOString(),
    pages: all.pages.sort((a, b) => a.route.localeCompare(b.route) || a.file.localeCompare(b.file)),
    httpCalls: all.httpCalls.sort(byLoc),
    routes: all.routes.sort(byLoc),
    dbAccess: dbAccess.sort(byLoc),
    tables: [...tables.values()].sort((a, b) => a.name.localeCompare(b.name)),
    models: all.models.sort((a, b) => a.model.localeCompare(b.model)),
    queues: all.queues.sort(byLoc),
  };
}

import { readFileSync } from "node:fs";
import { join, posix } from "node:path";
import ts from "typescript";

export interface Alias {
  prefix: string;
  targets: string[];
}

const EXTS = [".ts", ".tsx", ".js", ".jsx", ".mjs", ".cjs", ".mts", ".cts", ".vue", ".svelte"];

function readJsonLoose(path: string): Record<string, unknown> | null {
  try {
    const parsed = ts.parseConfigFileTextToJson(path, readFileSync(path, "utf8"));
    return (parsed.config as Record<string, unknown>) ?? null;
  } catch {
    return null;
  }
}

function firstString(value: unknown): string | undefined {
  if (typeof value === "string") return value;
  if (value && typeof value === "object") {
    for (const key of ["import", "node", "default", "require"]) {
      const v = (value as Record<string, unknown>)[key];
      const s = firstString(v);
      if (s) return s;
    }
  }
  return undefined;
}

export function loadAliases(repoAbs: string, files: string[]): Alias[] {
  const aliases: Alias[] = [];
  for (const rel of files) {
    const base = posix.basename(rel);
    const dir = posix.dirname(rel) === "." ? "" : posix.dirname(rel);
    if (base === "package.json") {
      const imports = (readJsonLoose(join(repoAbs, rel))?.imports ?? {}) as Record<string, unknown>;
      for (const [key, value] of Object.entries(imports)) {
        const target = firstString(value);
        if (!target || !key.endsWith("*") || !target.includes("*")) continue;
        aliases.push({ prefix: key.slice(0, -1), targets: [posix.normalize(posix.join(dir, target.slice(0, target.indexOf("*"))))] });
      }
    }
    if (base === "tsconfig.json" || base === "jsconfig.json") {
      const opts = (readJsonLoose(join(repoAbs, rel))?.compilerOptions ?? {}) as { baseUrl?: string; paths?: Record<string, string[]> };
      const baseUrl = posix.join(dir, opts.baseUrl ?? ".");
      for (const [key, targets] of Object.entries(opts.paths ?? {})) {
        if (!key.endsWith("*")) continue;
        aliases.push({
          prefix: key.slice(0, -1),
          targets: targets.filter((t) => t.includes("*")).map((t) => posix.normalize(posix.join(baseUrl, t.slice(0, t.indexOf("*"))))),
        });
      }
    }
  }
  return aliases;
}

function tryFile(base: string, files: Set<string>): string | null {
  const clean = base.replace(/^\.\//, "");
  const stem = clean.replace(/\.(js|jsx|mjs|cjs)$/, "");
  const candidates = [clean, ...EXTS.map((e) => stem + e), ...EXTS.map((e) => `${stem}/index${e}`)];
  return candidates.find((c) => files.has(c)) ?? null;
}

export function resolveImport(fromRel: string, source: string, files: Set<string>, aliases: Alias[]): string | null {
  const bases: string[] = [];
  if (source.startsWith("./") || source.startsWith("../")) {
    bases.push(posix.normalize(posix.join(posix.dirname(fromRel), source)));
  } else {
    for (const a of aliases) {
      if (source.startsWith(a.prefix)) bases.push(...a.targets.map((t) => posix.normalize(t + source.slice(a.prefix.length))));
    }
  }
  for (const b of bases) {
    const hit = tryFile(b, files);
    if (hit) return hit;
  }
  return null;
}

import { detectLayers, GraphBuilder, type KnowledgeGraph } from "@codeloom/core";
import { type Alias, resolveImport } from "./resolve-imports.js";
import type { FileEntry } from "./structure.js";

export function buildGraph(name: string, head: string, entries: Map<string, FileEntry>, aliases: Alias[]): KnowledgeGraph {
  const builder = new GraphBuilder(name, head);
  const files = new Set(entries.keys());
  for (const [rel, e] of entries) {
    builder.addFileWithAnalysis(rel, e.structure, { summary: "", fileSummary: "", summaries: {}, tags: [], complexity: "simple" });
  }

  const importsByFile = new Map<string, Array<{ target: string; specifiers: string[] }>>();
  for (const [rel, e] of entries) {
    const list: Array<{ target: string; specifiers: string[] }> = [];
    for (const imp of e.structure.imports) {
      const target = resolveImport(rel, imp.source, files, aliases);
      if (!target || target === rel) continue;
      builder.addImportEdge(rel, target);
      list.push({ target, specifiers: imp.specifiers });
    }
    importsByFile.set(rel, list);
  }

  const fnNames = new Map<string, Set<string>>();
  for (const [rel, e] of entries) fnNames.set(rel, new Set(e.structure.functions.map((f) => f.name)));

  for (const [rel, e] of entries) {
    const own = fnNames.get(rel)!;
    for (const call of e.calls) {
      if (!own.has(call.caller)) continue;
      const callee = call.callee.split(".").pop()!;
      if (own.has(callee)) {
        builder.addCallEdge(rel, call.caller, rel, callee);
        continue;
      }
      const imp = importsByFile.get(rel)!.find((i) => i.specifiers.includes(callee) && fnNames.get(i.target)?.has(callee));
      if (imp) builder.addCallEdge(rel, call.caller, imp.target, callee);
    }
  }

  const graph = builder.build();
  graph.layers = detectLayers(graph);
  return graph;
}

import { readFileSync, statSync } from "node:fs";
import { join } from "node:path";
import { contentHash, type KnowledgeGraph } from "@codeloom/core";
import { finalizeEndpoints, loadContext, runAdapters } from "../adapters/run.js";
import type { EndpointsPart } from "../adapters/types.js";
import { readJson, writeJsonAtomic } from "../store/json.js";
import { CODELOOM_DIR, FILES } from "../store/paths.js";
import {
  defaultRepoConfig, type EndpointsFile, type Fingerprints, FingerprintsSchema, RepoConfigSchema,
} from "../store/schemas.js";
import { listFiles } from "../workspace/files.js";
import { gitHead } from "../workspace/repos.js";
import { buildGraph } from "./build-graph.js";
import { loadAliases } from "./resolve-imports.js";
import { analyzeStructure, EMPTY_STRUCTURE, type FileEntry, getPlugin, isSourceFile } from "./structure.js";

const MAX_BYTES = 1_000_000;

export interface IndexStats {
  files: number;
  parsed: number;
  reused: number;
  skipped: Array<{ file: string; reason: string }>;
}

export interface IndexResult {
  graph: KnowledgeGraph;
  endpoints: EndpointsFile;
  stats: IndexStats;
}

type FileRecord = Fingerprints["files"][string] & FileEntry & { endpoints: EndpointsPart };

export async function indexRepo(repoAbs: string, repoId: string): Promise<IndexResult> {
  const dir = join(repoAbs, CODELOOM_DIR);
  const config = readJson(join(dir, FILES.config), RepoConfigSchema) ?? defaultRepoConfig();
  const files = listFiles(repoAbs, config.ignore);
  const ctx = loadContext(repoAbs, files);
  const contextKey = contentHash([...ctx.deps].sort().join(","));
  const cachePath = join(dir, FILES.fingerprints);
  const old = readJson(cachePath, FingerprintsSchema);
  const oldFiles = old && old.contextKey === contextKey ? old.files : {};

  const plugin = await getPlugin();
  const entries = new Map<string, FileEntry>();
  const parts: EndpointsPart[] = [];
  const next: Record<string, FileRecord> = {};
  const stats: IndexStats = { files: files.length, parsed: 0, reused: 0, skipped: [] };
  const empty = (rel: string) => entries.set(rel, { structure: EMPTY_STRUCTURE, calls: [] });

  for (const rel of files) {
    if (!isSourceFile(rel)) {
      empty(rel);
      continue;
    }
    const abs = join(repoAbs, rel);
    const size = statSync(abs).size;
    if (size > MAX_BYTES) {
      stats.skipped.push({ file: rel, reason: `larger than 1 MB (${size} bytes)` });
      empty(rel);
      continue;
    }
    const content = readFileSync(abs, "utf8");
    const hash = contentHash(content);
    const hit = oldFiles[rel] as FileRecord | undefined;
    let record: FileRecord;
    if (hit && hit.hash === hash) {
      record = hit;
      stats.reused++;
    } else {
      try {
        record = { hash, ...analyzeStructure(plugin, rel, content), endpoints: runAdapters(rel, content, ctx) };
        stats.parsed++;
      } catch (err) {
        stats.skipped.push({ file: rel, reason: err instanceof Error ? err.message : String(err) });
        empty(rel);
        continue;
      }
    }
    next[rel] = record;
    entries.set(rel, { structure: record.structure, calls: record.calls });
    parts.push(record.endpoints);
  }

  const graph = buildGraph(repoId, gitHead(repoAbs), entries, loadAliases(repoAbs, files));
  const endpoints = finalizeEndpoints(repoId, parts);
  writeJsonAtomic(join(dir, FILES.graph), graph);
  writeJsonAtomic(join(dir, FILES.endpoints), endpoints);
  writeJsonAtomic(cachePath, { version: 1, contextKey, files: next } satisfies Fingerprints);
  return { graph, endpoints, stats };
}

import { join } from "node:path";
import { type IndexStats, indexRepo } from "../indexer/index-repo.js";
import type { EndpointsFile } from "../store/schemas.js";
import { loadWorkspace } from "../workspace/repos.js";

export interface IndexRunResult {
  root: string;
  repos: Array<{ id: string; stats: IndexStats; endpoints: EndpointsFile }>;
}

export async function runIndex(start: string): Promise<IndexRunResult> {
  const ws = loadWorkspace(start);
  const repos: IndexRunResult["repos"] = [];
  for (const repo of ws.repos) {
    const res = await indexRepo(join(ws.root, repo.path), repo.id);
    repos.push({ id: repo.id, stats: res.stats, endpoints: res.endpoints });
  }
  return { root: ws.root, repos };
}

export function formatIndexSummary(r: IndexRunResult): string {
  const header = ["repo", "files", "parsed", "reused", "pages", "calls", "routes", "tables", "queues", "skipped"];
  const rows = r.repos.map((x) => [
    x.id,
    x.stats.files,
    x.stats.parsed,
    x.stats.reused,
    x.endpoints.pages.length,
    x.endpoints.httpCalls.length,
    x.endpoints.routes.length,
    x.endpoints.tables.length,
    x.endpoints.queues.length,
    x.stats.skipped.length,
  ].map(String));
  const widths = header.map((h, i) => Math.max(h.length, ...rows.map((row) => row[i]!.length)));
  const line = (cells: string[]) => cells.map((c, i) => (i === 0 ? c.padEnd(widths[i]!) : c.padStart(widths[i]!))).join("  ");
  const skipped = r.repos.flatMap((x) => x.stats.skipped.map((s) => `  ${x.id}/${s.file}: ${s.reason}`));
  return [
    line(header),
    ...rows.map(line),
    "",
    `${r.repos.length} repos indexed in ${r.root}`,
    ...(skipped.length ? ["Skipped files:", ...skipped] : []),
  ].join("\n");
}
```

import { chmodSync, existsSync, mkdirSync, readFileSync, statSync, writeFileSync } from "node:fs";
import { basename, join, posix, resolve } from "node:path";
import { CodeloomError } from "../errors.js";
import { writeJsonAtomic } from "../store/json.js";
import { CODELOOM_DIR, FILES, WORKSPACE_FILE } from "../store/paths.js";
import { defaultRepoConfig, type RepoRef, type WorkspaceFile } from "../store/schemas.js";
import { discoverRepos } from "../workspace/repos.js";

export const HOOK_MARK = "# codeloom: re-index after commit";
const CACHE_IGNORE = `${CODELOOM_DIR}/${FILES.fingerprints}`;

export interface InitResult {
  root: string;
  repos: RepoRef[];
  created: string[];
}

export function installHook(repoAbs: string, wsRoot: string): boolean {
  const gitDir = join(repoAbs, ".git");
  if (!existsSync(gitDir) || !statSync(gitDir).isDirectory()) return false;
  const hooksDir = join(gitDir, "hooks");
  mkdirSync(hooksDir, { recursive: true });
  const hook = join(hooksDir, "post-commit");
  const existing = existsSync(hook) ? readFileSync(hook, "utf8") : "#!/bin/sh\n";
  if (existing.includes(HOOK_MARK)) return false;
  const block = [
    HOOK_MARK,
    `if command -v codeloom >/dev/null 2>&1; then (cd ${JSON.stringify(wsRoot)} && codeloom index --quiet) >/dev/null 2>&1 & fi`,
  ].join("\n");
  writeFileSync(hook, `${existing.trimEnd()}\n\n${block}\n`);
  chmodSync(hook, 0o755);
  return true;
}

export function ensureGitignore(repoAbs: string): boolean {
  const path = join(repoAbs, ".gitignore");
  const current = existsSync(path) ? readFileSync(path, "utf8") : "";
  if (current.split(/\r?\n/).includes(CACHE_IGNORE)) return false;
  const prefix = current && !current.endsWith("\n") ? `${current}\n` : current;
  writeFileSync(path, `${prefix}${CACHE_IGNORE}\n`);
  return true;
}

export function runInit(dir: string): InitResult {
  const root = resolve(dir);
  const repos = discoverRepos(root);
  if (repos.length === 0) {
    throw new CodeloomError(
      `No git repos found in ${root}. Run "codeloom init" inside a repo, or in a folder that contains repos.`,
    );
  }
  const created: string[] = [];
  const isWorkspace = !(repos.length === 1 && repos[0]!.path === ".");
  if (isWorkspace && !existsSync(join(root, WORKSPACE_FILE))) {
    const ws: WorkspaceFile = { version: 1, name: basename(root), repos, links: [] };
    writeJsonAtomic(join(root, WORKSPACE_FILE), ws);
    created.push(WORKSPACE_FILE);
  }
  for (const repo of repos) {
    const repoAbs = join(root, repo.path);
    const config = join(repoAbs, CODELOOM_DIR, FILES.config);
    if (!existsSync(config)) {
      writeJsonAtomic(config, defaultRepoConfig());
      created.push(posix.join(repo.path, CODELOOM_DIR, FILES.config));
    }
    if (installHook(repoAbs, root)) created.push(posix.join(repo.path, ".git/hooks/post-commit"));
    if (ensureGitignore(repoAbs)) created.push(posix.join(repo.path, ".gitignore"));
  }
  return { root, repos, created };
}

export function formatInit(r: InitResult): string {
  const lines = [
    `Found ${r.repos.length} repo${r.repos.length === 1 ? "" : "s"} in ${r.root}: ${r.repos.map((x) => x.id).join(", ")}`,
    r.created.length ? "Created or updated:" : "Already set up; nothing changed.",
    ...r.created.map((c) => `  ${c}`),
    "",
    'Next: run "codeloom index" to build the map.',
  ];
  return lines.join("\n");
}
